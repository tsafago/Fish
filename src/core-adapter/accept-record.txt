    public void attachOutput(java.io.OutputStream output) {
        bout = new java.io.BufferedOutputStream(output);
        connected = true;
        sfParser = new WTDSFParser(this);
        bk = new Stream5250();
    }

    private boolean rejectUnsupportedDisplayField() {
        throw new IllegalStateException("Unsupported enhanced display structured field; session stopped to preserve synchronization.");
    }

    private boolean noticeReadOpen, noticeInvited;
    private boolean noticeParsed, noticeDisplayChanged, noticeReadInRecord;

    public void discardStartupNoticeInput() {
        noticeReadOpen = noticeInvited = false;
    }

    public boolean startupNoticeInputReady() {
        return noticeReadOpen && noticeInvited;
    }

    public String startupNoticeSequenceState() {
        if (!noticeReadOpen) return "waiting for a host input-read record for this display";
        return noticeInvited ? "host input read and invite received" : "waiting for a host invite record";
    }

    public boolean acceptRecord(byte[] record) {
        if (!javafx.application.Platform.isFxApplicationThread())
            throw new IllegalStateException("Core records must be processed on the JavaFX Application Thread.");
        bk.initialize(record);
        noticeParsed = noticeDisplayChanged = noticeReadInRecord = false;
        screen52.setCursorActive(false);
        switch (bk.getOpCode()) {
            case 0: break;
            case 1:
                parseIncoming();
                pendingUnlock = true;
                cursorOn = true;
                setInvited();
                break;
            case 2:
                parseIncoming();
                break;
            case 3:
                parseIncoming();
                setInvited();
                break;
            case 4:
            case 5:
                parseIncoming();
                break;
            case 6:
                sendAidKey(0);
                break;
            case 8:
                try { readScreen(); }
                catch (java.io.IOException e) { throw new IllegalStateException("Read screen reply failed", e); }
                break;
            case 10:
                cancelInvite();
                break;
            case 11:
                screen52.getOIA().setMessageLightOn();
                break;
            case 12:
                screen52.getOIA().setMessageLightOff();
                break;
            default:
                throw new IllegalStateException("Unsupported GDS opcode: " + bk.getOpCode());
        }
        if (strpccmd)
            throw new IllegalStateException("Host STRPCCMD execution is disabled.");
        screen52.updateDirty();
        if (pendingUnlock && !screen52.isStatusErrorCode()) {
            screen52.getOIA().setKeyBoardLocked(false);
            pendingUnlock = false;
        }
        if (cursorOn && !screen52.getOIA().isKeyBoardLocked()) {
            screen52.setCursorActive(true);
            cursorOn = false;
        }
        int opcode = bk.getOpCode();
        boolean invited = opcode == 1 || opcode == 3;
        boolean complete = bk.streamSize == record.length && bk.getCurrentPos() == bk.streamSize;
        if (complete && noticeParsed && (invited || opcode == 2)) {
            // A fresh read commits the current buffer, regardless of how the host built it.
            if (noticeDisplayChanged || noticeReadInRecord) noticeReadOpen = noticeReadInRecord;
            if (invited) noticeInvited = true;
        } else if (!(complete && bk.dataStart == bk.streamSize && (opcode == 11 || opcode == 12))) {
            discardStartupNoticeInput();
        }
        return startupNoticeInputReady();
    }
